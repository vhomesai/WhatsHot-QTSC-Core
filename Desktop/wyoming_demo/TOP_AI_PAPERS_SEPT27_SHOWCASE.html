class SIFTTournamentEvaluator:
    """Implements Bradley-Terry pairwise tournament selection to conserve Heron QPU time."""
    def solve_bradley_terry(self, win_matrix, max_iter=100, tol=1e-4):
        N = win_matrix.shape[0]
        theta = np.zeros(N)
        for _ in range(max_iter):
            old_theta = theta.copy()
            for i in range(N):
                denom = sum(np.exp(old_theta[j]) / (np.exp(old_theta[i]) + np.exp(old_theta[j])) 
                            for j in range(N) if i != j)
                theta[i] = np.log(max(sum(win_matrix[i, :]), 1e-3)) - np.log(max(denom, 1e-3))
            theta -= np.mean(theta)  # Regularization zero-mean anchor
        return theta
